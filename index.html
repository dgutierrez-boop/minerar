function actualizarResumenEjecutivo() { const sucursal = document.getElementById('ejecutivoSucursal').value; const mes = document.getElementById('ejecutivoMes').value; const sector = document.getElementById('ejecutivoSector').value; const estadoFiltro = document.getElementById('ejecutivoEstadoFiltro').value; const tipoFiltro = document.getElementById('ejecutivoTipoFiltro').value; console.log(`🔍 Actualizando resumen: Sucursal=${sucursal}, Mes=${mes}`); let datosActos = [], datosVCP = []; // Recorrer TODAS las sucursales si está seleccionado "todas" const sucursalesAActualizar = (sucursal === 'todas') ? ['MALAGUENO', 'BARKER', 'LAPAMPITA'] : [sucursal]; sucursalesAActualizar.forEach(suc => { const actosSucursal = ALL_ACTOS_DATA[suc] || []; const vcpSucursal = ALL_VCP_DATA[suc] || []; console.log(` ${suc}: ${actosSucursal.length} actos/cond, ${vcpSucursal.length} VCP`); datosActos = datosActos.concat(actosSucursal.filter(d => (mes === 'todo' || d.mes === mes) && (sector === 'todos' || d.sector.includes(sector)) && (estadoFiltro === 'todos' || (estadoFiltro === 'abiertos' ? !d.cerrado : d.cerrado)) && (tipoFiltro === 'todos' || tipoFiltro === 'vcp' || d.tipo === tipoFiltro) )); if (vcpSucursal.length > 0) { datosVCP = datosVCP.concat(vcpSucursal.filter(d => (mes === 'todo' || d.mes === mes) && (sector === 'todos' || d.sector.includes(sector)) && (estadoFiltro === 'todos' || (estadoFiltro === 'abiertos' ? !d.cerrado : d.cerrado)) && (tipoFiltro === 'todos' || tipoFiltro === 'vcp') )); } }); // Aplicar filtro adicional por tipo si es necesario if (currentFiltroTipo !== 'todos') { if (currentFiltroTipo === 'acto') datosVCP = []; else if (currentFiltroTipo === 'condicion') { datosActos = datosActos.filter(d => d.tipo === 'condicion'); datosVCP = []; } else if (currentFiltroTipo === 'vcp') datosActos = []; } const totalActos = datosActos.filter(d => d.tipo === 'acto').length; const totalCond = datosActos.filter(d => d.tipo === 'condicion').length; const totalVCP = datosVCP.length; console.log(`📊 RESULTADO: Actos=${totalActos}, Cond=${totalCond}, VCP=${totalVCP}, Total=${totalActos+totalCond+totalVCP}`); document.getElementById('ejecTotalActos').innerText = totalActos; document.getElementById('ejecTotalCondiciones').innerText = totalCond; document.getElementById('ejecTotalVCP').innerText = totalVCP; document.getElementById('ejecTotalGeneral').innerText = totalActos + totalCond + totalVCP; renderEjecutivoRankings(datosActos); renderEjecutivoReincidencias(datosActos); renderEjecutivoTabla([...datosActos, ...datosVCP]); renderEjecutivoChartsSucursales(sector, sucursal, estadoFiltro); }
